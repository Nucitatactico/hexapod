#define PCA9685_ADDR     0x40
#define PCA9685_MODE1    0x00
#define PCA9685_PRESCALE 0xFE

static i2c_master_bus_handle_t i2c_bus = NULL;
static i2c_master_dev_handle_t pca_dev = NULL;

float ang_actual[12];
int   ang_objetivo[12];

#define INTERVALO_SUAVIZADO_MS 15
#define VEL_SUAVIZADO_COXA     1.5f
#define VEL_SUAVIZADO_FEMUR    4.0f

static esp_err_t pca_write_reg(uint8_t reg, uint8_t val) {
    uint8_t buf[2] = { reg, val };
    return i2c_master_transmit(pca_dev, buf, 2, 1000);
}

static void pca9685_set_pwm_freq(float freq_hz) {
    uint8_t prescale = (uint8_t)((25000000.0f / (4096.0f * freq_hz)) - 1.0f + 0.5f);
    pca_write_reg(PCA9685_MODE1, 0x10);
    pca_write_reg(PCA9685_PRESCALE, prescale);
    vTaskDelay(pdMS_TO_TICKS(1));
    pca_write_reg(PCA9685_MODE1, 0x00);
    vTaskDelay(pdMS_TO_TICKS(1));
}

static void pca9685_set_channel(uint8_t canal, uint16_t valor) {
    uint8_t base = 0x06 + 4 * canal;
    uint8_t off_l = valor & 0xFF;
    uint8_t off_h = (valor >> 8) & 0x0F;
    pca_write_reg(base + 0, 0);
    pca_write_reg(base + 1, 0);
    pca_write_reg(base + 2, off_l);
    pca_write_reg(base + 3, off_h);
}

int angulo_a_pulsos(int angulo) {
    if (angulo < 0)   angulo = 0;
    if (angulo > 180) angulo = 180;
    return 130 + (angulo * (480 - 130) / 180);
}

void mover_servo_inmediato(uint8_t pin, int angulo) {
    if (pin > 11) return;
    ang_actual[pin]   = (float)angulo;
    ang_objetivo[pin] = angulo;
    pca9685_set_channel(pin, angulo_a_pulsos(angulo));
}

void mover_servo(uint8_t pin, int angulo) {
    if (pin > 11) return;
    ang_objetivo[pin] = angulo;
}

void procesar_movimiento_suave(void) {
    for (int i = 0; i < 12; i++) {
        float diff = (float)ang_objetivo[i] - ang_actual[i];
        if (fabsf(diff) > 0.5f) {
            float vel = (i % 2 == 1) ? VEL_SUAVIZADO_FEMUR : VEL_SUAVIZADO_COXA;
            if (diff > 0.0f) ang_actual[i] += vel;
            else             ang_actual[i] -= vel;
            if (fabsf((float)ang_objetivo[i] - ang_actual[i]) <= vel) {
                ang_actual[i] = (float)ang_objetivo[i];
            }
            pca9685_set_channel(i, angulo_a_pulsos((int)ang_actual[i]));
        }
    }
}

void mover_todas_caderas_centro(void) {
    mover_servo_inmediato(D_IZQ_C, COXA_CENTRO);
    mover_servo_inmediato(M_IZQ_C, COXA_CENTRO);
    mover_servo_inmediato(P_IZQ_C, COXA_CENTRO);
    mover_servo_inmediato(D_DER_C, COXA_CENTRO);
    mover_servo_inmediato(M_DER_C, COXA_CENTRO);
    mover_servo_inmediato(P_DER_C, COXA_CENTRO);
}

void mover_tripode_a(int femur, int coxa_izq, int coxa_der) {
    mover_servo(D_IZQ_F, femur); mover_servo(D_IZQ_C, coxa_izq);
    mover_servo(P_IZQ_F, femur); mover_servo(P_IZQ_C, coxa_izq);
    mover_servo(M_DER_F, femur); mover_servo(M_DER_C, coxa_der);
}

void mover_tripode_b(int femur, int coxa_izq, int coxa_der) {
    mover_servo(M_IZQ_F, femur); mover_servo(M_IZQ_C, coxa_izq);
    mover_servo(D_DER_F, femur); mover_servo(D_DER_C, coxa_der);
    mover_servo(P_DER_F, femur); mover_servo(P_DER_C, coxa_der);
}

esp_err_t hardware_init(void) {
    gpio_config_t io_cfg = {
        .pin_bit_mask = (1ULL << PIN_RADAR) | (1ULL << PIN_CDS),
        .mode         = GPIO_MODE_INPUT_OUTPUT,
        .pull_up_en   = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type    = GPIO_INTR_DISABLE,
    };
    gpio_config(&io_cfg);
    gpio_set_level(PIN_CDS, 1);

    i2c_master_bus_config_t bus_cfg = {
        .clk_source        = I2C_CLK_SRC_DEFAULT,
        .i2c_port          = I2C_NUM_0,
        .scl_io_num        = PIN_SCL,
        .sda_io_num        = PIN_SDA,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_cfg, &i2c_bus));

    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address  = PCA9685_ADDR,
        .scl_speed_hz    = 400000,
    };
    ESP_ERROR_CHECK(i2c_master_bus_add_device(i2c_bus, &dev_cfg, &pca_dev));

    pca_write_reg(PCA9685_MODE1, 0x00);
    pca9685_set_pwm_freq(50.0f);

    for (int i = 0; i < 12; i++) {
        int a = (i % 2 == 1) ? FEMUR_ACOSTADO : COXA_CENTRO;
        mover_servo_inmediato(i, a);
    }

    ESP_LOGI(TAG, "Hardware listo (PCA9685 en 0x40, I2C 7/8)");
    return ESP_OK;
}

static const char *TAG = "MOTION";

bool robot_encendido              = false;
bool modo_automatico              = false;
bool modo_calibracion             = false;
bool radar_fisicamente_encendido  = true;
int  direccion_actual             = 0;   // 0=stop, 1=go, 2=back, 4=right, 5=left
int  velocidad_caminata           = 600;
bool en_transicion                = false;

static char accion_actual[32] = "ESTATICO";

void set_accion(const char *texto) {
    strncpy(accion_actual, texto, sizeof(accion_actual) - 1);
    accion_actual[sizeof(accion_actual) - 1] = '\0';
}
const char *get_accion(void) { return accion_actual; }

static int fase_marcha = 0;
static unsigned long tiempo_paso = 0;
static int direccion_anterior = 0;

// ─── Evasión ───
static bool modo_evasion = false;
static int direccion_evasion = 0;
static TickType_t t_evasion_ini = 0;

// ─── Helpers de trípode con amplitudes por pata ───
// Cada función recibe 6 amplitudes (grados desde COXA_CENTRO) para cada pata:
// orden: D_IZQ, M_IZQ, P_IZQ, D_DER, M_DER, P_DER
// La función suma esas amplitudes a COXA_CENTRO con el signo adecuado
// para que "positivo" siempre signifique "adelante" en esa pata.

static void mover_tripode_a_amp(int amp_d_izq, int amp_m_izq, int amp_p_izq,
                                 int amp_d_der, int amp_m_der, int amp_p_der) {
    // Trípode A: D_IZQ, P_IZQ, M_DER
    mover_servo(D_IZQ_C, COXA_CENTRO + amp_d_izq);
    mover_servo(P_IZQ_C, COXA_CENTRO + amp_p_izq);
    mover_servo(M_DER_C, COXA_CENTRO - amp_m_der);   // derecha: signo invertido
}

static void mover_tripode_b_amp(int amp_d_izq, int amp_m_izq, int amp_p_izq,
                                 int amp_d_der, int amp_m_der, int amp_p_der) {
    // Trípode B: M_IZQ, D_DER, P_DER
    mover_servo(M_IZQ_C, COXA_CENTRO + amp_m_izq);
    mover_servo(D_DER_C, COXA_CENTRO - amp_d_der);
    mover_servo(P_DER_C, COXA_CENTRO - amp_p_der);
}

// ─── Marcha de 6 fases con amplitudes por pata ───
static void caminar_amp(int direccion,
                        int f_d_izq, int f_m_izq, int f_p_izq,
                        int f_d_der, int f_m_der, int f_p_der,
                        int b_d_izq, int b_m_izq, int b_p_izq,
                        int b_d_der, int b_m_der, int b_p_der) {
    if ((direccion == 1 || direccion == 2) &&
        radar_fisicamente_encendido && gpio_get_level(PIN_RADAR)) {
        return;
    }
    unsigned long ahora = xTaskGetTickCount() * portTICK_PERIOD_MS;
    if (ahora - tiempo_paso < (unsigned long)velocidad_caminata) return;
    tiempo_paso = ahora;

    // Fase 0: A center-up, B fija en back-down
    // Fase 1: A forward-up, B back-down
    // Fase 2: A forward-down (planta)
    // Fase 3: B center-up, A fija en back-down
    // Fase 4: B forward-up, A back-down
    // Fase 5: B forward-down (planta)
    if (fase_marcha == 0) {
        mover_tripode_a_amp(0, 0, 0, 0, 0, 0);
        // fémures A arriba
        mover_servo(D_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(P_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(M_DER_F, FEMUR_LEVANTADO);
        mover_tripode_b_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(M_IZQ_F, FEMUR_APOYO);
        mover_servo(D_DER_F, FEMUR_APOYO);
        mover_servo(P_DER_F, FEMUR_APOYO);
    }
    else if (fase_marcha == 1) {
        mover_tripode_a_amp(f_d_izq, f_m_izq, f_p_izq, f_d_der, f_m_der, f_p_der);
        mover_servo(D_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(P_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(M_DER_F, FEMUR_LEVANTADO);
        mover_tripode_b_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(M_IZQ_F, FEMUR_APOYO);
        mover_servo(D_DER_F, FEMUR_APOYO);
        mover_servo(P_DER_F, FEMUR_APOYO);
    }
    else if (fase_marcha == 2) {
        mover_tripode_a_amp(f_d_izq, f_m_izq, f_p_izq, f_d_der, f_m_der, f_p_der);
        mover_servo(D_IZQ_F, FEMUR_APOYO);
        mover_servo(P_IZQ_F, FEMUR_APOYO);
        mover_servo(M_DER_F, FEMUR_APOYO);
        mover_tripode_b_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(M_IZQ_F, FEMUR_APOYO);
        mover_servo(D_DER_F, FEMUR_APOYO);
        mover_servo(P_DER_F, FEMUR_APOYO);
    }
    else if (fase_marcha == 3) {
        mover_tripode_a_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(D_IZQ_F, FEMUR_APOYO);
        mover_servo(P_IZQ_F, FEMUR_APOYO);
        mover_servo(M_DER_F, FEMUR_APOYO);
        mover_tripode_b_amp(0, 0, 0, 0, 0, 0);
        mover_servo(M_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(D_DER_F, FEMUR_LEVANTADO);
        mover_servo(P_DER_F, FEMUR_LEVANTADO);
    }
    else if (fase_marcha == 4) {
        mover_tripode_a_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(D_IZQ_F, FEMUR_APOYO);
        mover_servo(P_IZQ_F, FEMUR_APOYO);
        mover_servo(M_DER_F, FEMUR_APOYO);
        mover_tripode_b_amp(f_d_izq, f_m_izq, f_p_izq, f_d_der, f_m_der, f_p_der);
        mover_servo(M_IZQ_F, FEMUR_LEVANTADO);
        mover_servo(D_DER_F, FEMUR_LEVANTADO);
        mover_servo(P_DER_F, FEMUR_LEVANTADO);
    }
    else if (fase_marcha == 5) {
        mover_tripode_a_amp(b_d_izq, b_m_izq, b_p_izq, b_d_der, b_m_der, b_p_der);
        mover_servo(D_IZQ_F, FEMUR_APOYO);
        mover_servo(P_IZQ_F, FEMUR_APOYO);
        mover_servo(M_DER_F, FEMUR_APOYO);
        mover_tripode_b_amp(f_d_izq, f_m_izq, f_p_izq, f_d_der, f_m_der, f_p_der);
        mover_servo(M_IZQ_F, FEMUR_APOYO);
        mover_servo(D_DER_F, FEMUR_APOYO);
        mover_servo(P_DER_F, FEMUR_APOYO);
    }

    fase_marcha = (fase_marcha + 1) % 6;
}

// ─── Amplitudes para cada dirección ───
// forward/back: ángulo 25° (≈ 115°-90°)
// strafe: patas laterales 25°, diagonales 10° (atenuadas para no rotar)

#define AMP_FWD   25
#define AMP_BACK -25
#define AMP_STRAFE_LAT  25
#define AMP_STRAFE_DIAG 10

// GO (adelante): izquierdas adelante, derechas adelante (en su marco invertido)
static void caminar_go(void) {
    set_accion("MOV: ADELANTE");
    caminar_amp(1,
        AMP_FWD, AMP_FWD, AMP_FWD,   // A fwd
        AMP_FWD, AMP_FWD, AMP_FWD,
        AMP_BACK, AMP_BACK, AMP_BACK, // B back
        AMP_BACK, AMP_BACK, AMP_BACK);
}

// BACK (atrás): izquierdas atrás, derechas atrás
static void caminar_back(void) {
    set_accion("MOV: ATRAS");
    caminar_amp(2,
        AMP_BACK, AMP_BACK, AMP_BACK,
        AMP_BACK, AMP_BACK, AMP_BACK,
        AMP_FWD, AMP_FWD, AMP_FWD,
        AMP_FWD, AMP_FWD, AMP_FWD);
}

// LEFT (strafe izquierda):
// - izq laterales empujan "adelante" (en su marco, eso lleva el cuerpo a la izq)
// - izq diagonales empujan "atrás" atenuado (para cancelar rotación)
// - der laterales empujan "adelante" (en su marco invertido, eso también lleva el cuerpo a la izq)
// - der diagonales empujan "atrás" atenuado
static void caminar_left(void) {
    set_accion("STRAFE: IZQ");
    caminar_amp(5,
        // A fwd (patas que empujan)
        -AMP_STRAFE_DIAG, AMP_STRAFE_LAT, -AMP_STRAFE_DIAG,   // D_IZQ, M_IZQ, P_IZQ
         AMP_STRAFE_DIAG, AMP_STRAFE_LAT,  AMP_STRAFE_DIAG,   // D_DER, M_DER, P_DER
        // B back (patas que recuperan)
         AMP_STRAFE_DIAG, -AMP_STRAFE_LAT,  AMP_STRAFE_DIAG,
        -AMP_STRAFE_DIAG, -AMP_STRAFE_LAT, -AMP_STRAFE_DIAG);
}

// RIGHT (strafe derecha): espejo de LEFT
static void caminar_right(void) {
    set_accion("STRAFE: DER");
    caminar_amp(4,
        // A fwd
         AMP_STRAFE_DIAG, -AMP_STRAFE_LAT,  AMP_STRAFE_DIAG,
        -AMP_STRAFE_DIAG, -AMP_STRAFE_LAT, -AMP_STRAFE_DIAG,
        // B back
        -AMP_STRAFE_DIAG, AMP_STRAFE_LAT, -AMP_STRAFE_DIAG,
         AMP_STRAFE_DIAG, AMP_STRAFE_LAT,  AMP_STRAFE_DIAG);
}

// ─── Dispatcher ───
static void caminar(int direccion) {
    switch (direccion) {
        case 1: caminar_go();    break;
        case 2: caminar_back();  break;
        case 4: caminar_right(); break;
        case 5: caminar_left();  break;
    }
}

// ─── Transición postura lenta (2 s) ───
void transicion_postura(bool levantar) {
    en_transicion = true;
    set_accion(levantar ? "LEVANTANDO..." : "APAGANDO...");

    int inicial = levantar ? FEMUR_ACOSTADO : FEMUR_APOYO;
    int final_  = levantar ? FEMUR_APOYO    : FEMUR_ACOSTADO;
    int inc     = (final_ > inicial) ? 5 : -5;
    int actual  = inicial;

    mover_todas_caderas_centro();
    vTaskDelay(pdMS_TO_TICKS(200));

    while (actual != final_) {
        actual += inc;
        if ((inc > 0 && actual > final_) || (inc < 0 && actual < final_)) actual = final_;

        mover_servo_inmediato(D_IZQ_F, actual);
        mover_servo_inmediato(M_IZQ_F, actual);
        mover_servo_inmediato(P_IZQ_F, actual);
        mover_servo_inmediato(D_DER_F, actual);
        mover_servo_inmediato(M_DER_F, actual);
        mover_servo_inmediato(P_DER_F, actual);
        vTaskDelay(pdMS_TO_TICKS(100));
    }

    robot_encendido = levantar;
    set_accion("ESTATICO");
    en_transicion = false;
}

// ─── Flexiones ───
void flexiones(void) {
    en_transicion = true;
    set_accion("FLEXIONES");

    for (int r = 0; r < 3; r++) {
        for (int i = 1; i < 12; i += 2) mover_servo_inmediato(i, FEMUR_APOYO);
        mover_todas_caderas_centro();
        vTaskDelay(pdMS_TO_TICKS(400));

        for (int i = 1; i < 12; i += 2) mover_servo_inmediato(i, FEMUR_ACOSTADO);
        vTaskDelay(pdMS_TO_TICKS(400));
    }

    mover_tripode_a(FEMUR_APOYO, COXA_CENTRO, COXA_CENTRO);
    mover_tripode_b(FEMUR_APOYO, COXA_CENTRO, COXA_CENTRO);
    set_accion("ESTATICO");
    en_transicion = false;
}

// ─── Tarea principal con evasión ───
void motion_task(void *pvParameters) {
    while (1) {
        if (!en_transicion) {
            procesar_movimiento_suave();
        }

        if (robot_encendido && !en_transicion) {
            bool hay_obstaculo = radar_fisicamente_encendido && gpio_get_level(PIN_RADAR);
            TickType_t ahora = xTaskGetTickCount();

            // Iniciar evasión
            if (hay_obstaculo && (direccion_actual == 1 || direccion_actual == 2) && !modo_evasion) {
                modo_evasion = true;
                direccion_evasion = direccion_actual;
                t_evasion_ini = ahora;
                direccion_actual = 4;   // giro derecha (reusa strafe_right)
                fase_marcha = 0;
                set_accion("EVADIENDO: DER");
                ESP_LOGI(TAG, "Obstáculo detectado, evadiendo a la derecha");
            }

            if (modo_evasion) {
                uint32_t elapsed_ms = (ahora - t_evasion_ini) * portTICK_PERIOD_MS;
                if ((elapsed_ms > 1500 && !hay_obstaculo) || elapsed_ms > 4000) {
                    modo_evasion = false;
                    direccion_actual = direccion_evasion;
                    fase_marcha = 0;
                    ESP_LOGI(TAG, "Retomando dirección original");
                }
            }

            if (!modo_calibracion) {
                if (direccion_actual != direccion_anterior) {
                    fase_marcha = 0;
                    direccion_anterior = direccion_actual;
                }

                if (direccion_actual != 0) {
                    caminar(direccion_actual);
                } else {
                    mover_tripode_a(FEMUR_APOYO, COXA_CENTRO, COXA_CENTRO);
                    mover_tripode_b(FEMUR_APOYO, COXA_CENTRO, COXA_CENTRO);
                    fase_marcha = 0;
                    if (strcmp(accion_actual, "FRENO X OBSTACULO") != 0) {
                        set_accion("ESTATICO");
                    }
                }
            } else {
                set_accion("MODO SERVOS");
            }
        }

        vTaskDelay(pdMS_TO_TICKS(20));
    }
}

// Pines del micrófono PDM en la FireBeetle 2 ESP32-P4
#define PDM_CLK_PIN   12
#define PDM_DATA_PIN  9

static const esp_afe_sr_iface_t *afe_handle = NULL;
static esp_afe_sr_data_t        *afe_data   = NULL;
static esp_mn_iface_t           *multinet   = NULL;
static model_iface_data_t       *mn_data    = NULL;
static bool voice_ready = false;

static i2s_chan_handle_t rx_handle = NULL;

static int ultimo_id = -1;
static TickType_t tiempo_ultimo_id = 0;
#define DEBOUNCE_MS 800

static void aplicar_comando_voz(int id) {
	TickType_t ahora = xTaskGetTickCount() * portTICK_PERIOD_MS;
	if (id == ultimo_id && (ahora - tiempo_ultimo_id) < DEBOUNCE_MS) return;
	ultimo_id = id;
	tiempo_ultimo_id = ahora;

	ESP_LOGI(TAG, "Comando de voz detectado: id=%d", id);

	switch (id) {
		case 0: // "go"
			if (!modo_automatico && robot_encendido && !modo_calibracion) {
				if (radar_fisicamente_encendido && gpio_get_level(PIN_RADAR)) {
					direccion_actual = 0;
					set_accion("FRENO X OBSTACULO");
				} else {
					direccion_actual = 1;   // forward
				}
			}
			break;
		case 1: // "back"
			if (!modo_automatico && robot_encendido && !modo_calibracion) {
				if (radar_fisicamente_encendido && gpio_get_level(PIN_RADAR)) {
					direccion_actual = 0;
					set_accion("FRENO X OBSTACULO");
				} else {
					direccion_actual = 2;   // backward
				}
			}
			break;
		case 2: // "left"
			if (!modo_automatico && robot_encendido && !modo_calibracion)
				direccion_actual = 3;   // strafe left
			break;
		case 3: // "right"
			if (!modo_automatico && robot_encendido && !modo_calibracion)
				direccion_actual = 4;   // strafe right
			break;
		case 4: // "jump"
			if (robot_encendido && !modo_automatico && !modo_calibracion)
				flexiones();
			break;
		case 5: // "power"
			transicion_postura(!robot_encendido);
			if (!robot_encendido) {
				direccion_actual = 0;
				modo_calibracion = false;
			}
			break;
	}
}

esp_err_t voice_init(void) {
	// 1. Modelos desde la partición
	srmodel_list_t *models = esp_srmodel_init("model");
	if (!models) {
		ESP_LOGE(TAG, "esp_srmodel_init falló");
		return ESP_FAIL;
	}

	// 2. AFE: "M" = 1 micrófono (mono, sin referencia de playback)
	afe_config_t *afe_config = afe_config_init("M", models, AFE_TYPE_SR, AFE_MODE_HIGH_PERF);
	if (!afe_config) {
		ESP_LOGE(TAG, "afe_config_init falló");
		return ESP_FAIL;
	}

	afe_handle = esp_afe_handle_from_config(afe_config);
	if (!afe_handle) {
		ESP_LOGE(TAG, "afe_handle falló");
		return ESP_FAIL;
	}

	afe_data = afe_handle->create_from_config(afe_config);
	if (!afe_data) {
		ESP_LOGE(TAG, "afe create falló");
		return ESP_FAIL;
	}

	// 3. MultiNet
	char *mn_name = esp_srmodel_filter(models, ESP_MN_PREFIX, ESP_MN_ENGLISH);
	if (!mn_name) {
		ESP_LOGE(TAG, "no MN inglés");
		return ESP_FAIL;
	}
	ESP_LOGI(TAG, "Modelo MultiNet: %s", mn_name);

	multinet = esp_mn_handle_from_name(mn_name);
	if (!multinet) {
		ESP_LOGE(TAG, "mn_handle falló");
		return ESP_FAIL;
	}

	mn_data = multinet->create(mn_name, 5760);
	if (!mn_data) {
		ESP_LOGE(TAG, "mn create falló");
		return ESP_FAIL;
	}

	// 4. Comandos
	esp_mn_commands_clear();
	esp_mn_commands_add(0, "go");
	esp_mn_commands_add(1, "back");
	esp_mn_commands_add(2, "left");
	esp_mn_commands_add(3, "right");
	esp_mn_commands_add(4, "jump");
	esp_mn_commands_add(5, "power");
	esp_mn_commands_update();

	// 5. I2S PDM RX (micrófono de la FireBeetle 2)
	i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(I2S_NUM_0, I2S_ROLE_MASTER);
	ESP_ERROR_CHECK(i2s_new_channel(&chan_cfg, NULL, &rx_handle));

	i2s_pdm_rx_config_t pdm_cfg = {
		.clk_cfg  = I2S_PDM_RX_CLK_DEFAULT_CONFIG(16000),
		.slot_cfg = I2S_PDM_RX_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO),
		.gpio_cfg = {
			.clk = PDM_CLK_PIN,
			.din = PDM_DATA_PIN,
		},
	};
	ESP_ERROR_CHECK(i2s_channel_init_pdm_rx_mode(rx_handle, &pdm_cfg));
	ESP_ERROR_CHECK(i2s_channel_enable(rx_handle));

	voice_ready = true;
	ESP_LOGI(TAG, "ESP-SR inicializado (micrófono PDM GPIO12/9, mono)");
	return ESP_OK;
}

// ─────────────────────────────────────────────────────────────
// Tarea unificada: lee I2S → feed → fetch → detect
// ─────────────────────────────────────────────────────────────
void mic_feed_task(void *pvParameters) {
	if (!voice_ready || !rx_handle) {
		vTaskDelete(NULL);
		return;
	}

	int chunksize = afe_handle->get_feed_chunksize(afe_data);
	size_t size   = (size_t)chunksize * sizeof(int16_t);

	int16_t *buffer = malloc(size);
	if (!buffer) {
		vTaskDelete(NULL);
		return;
	}

	ESP_LOGI(TAG, "mic_feed_task (chunk=%d, bytes=%u)", chunksize, (unsigned)size);

	// Para no inundar el log, imprimimos el peak cada N ciclos
	int contador_peak = 0;
	int peak_max = 0;

	while (1) {
		size_t bytes_read = 0;
		esp_err_t ret = i2s_channel_read(rx_handle, buffer, size, &bytes_read, portMAX_DELAY);
		if (ret != ESP_OK || bytes_read == 0) {
			vTaskDelay(pdMS_TO_TICKS(10));
			continue;
		}

		// ── Diagnóstico: calcular peak del buffer de audio ──
		int peak = 0;
		int muestras = bytes_read / sizeof(int16_t);
		for (int i = 0; i < muestras; i++) {
			int v = buffer[i];
			if (v < 0) v = -v;
			if (v > peak) peak = v;
		}
		if (peak > peak_max) peak_max = peak;
		contador_peak++;
		if (contador_peak >= 50) {  // ~cada 1 segundo (50 chunks × 20ms)
			ESP_LOGI(TAG, "audio peak (1s) = %d", peak_max);
			contador_peak = 0;
			peak_max = 0;
		}

		// 1. Alimentar el AFE
		afe_handle->feed(afe_data, buffer);

		// 2. Extraer el resultado inmediatamente
		afe_fetch_result_t *res = afe_handle->fetch(afe_data);
		if (!res || res->ret_value == ESP_FAIL) {
			continue;
		}

		// ── Diagnóstico: log de wake word ──
		if (res->wakeup_state == WAKENET_DETECTED) {
			ESP_LOGI(TAG, ">>> WAKE WORD DETECTADA <<<");
		}

		// 3. Pasar al MultiNet
		esp_mn_state_t state = multinet->detect(mn_data, res->data);
		if (state == ESP_MN_STATE_DETECTED) {
			esp_mn_results_t *r = multinet->get_results(mn_data);
			if (r && r->num > 0) {
				aplicar_comando_voz(r->phrase_id[0]);
			}
		}
	}
}
